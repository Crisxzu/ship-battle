package com.par_28.ship_battle.examples;

import com.par_28.ship_battle.model.*;
import com.par_28.ship_battle.model.ai.AIPlayer;
import com.par_28.ship_battle.model.ai.enums.AIDifficulty;
import com.par_28.ship_battle.model.enums.Direction;

/**
 * Example demonstrating how to use the AI system in a game.
 * <p>
 * This example shows:
 * - Creating an AI player with different difficulty levels
 * - Setting up a game between human and AI
 * - Handling AI turns
 * - Getting AI shot decisions
 * </p>
 */
public class AIGameExample {

    /**
     * Example 1: Create AI players with different difficulties
     */
    public static void exampleCreateAIPlayers() {
        System.out.println("=== Example 1: Creating AI Players ===\n");

        // Easy AI - Random shots
        AIPlayer easyAI = new AIPlayer("Easy Bot", 10, AIDifficulty.EASY);
        System.out.println("Created: " + easyAI.getName() + " (Difficulty: " + easyAI.getDifficulty() + ")");

        // Medium AI - Hunt and target
        AIPlayer mediumAI = new AIPlayer("Medium Bot", 10, AIDifficulty.MEDIUM);
        System.out.println("Created: " + mediumAI.getName() + " (Difficulty: " + mediumAI.getDifficulty() + ")");

        // Hard AI - Probability-based
        AIPlayer hardAI = new AIPlayer("Hard Bot", 10, AIDifficulty.HARD);
        System.out.println("Created: " + hardAI.getName() + " (Difficulty: " + hardAI.getDifficulty() + ")");

        System.out.println();
    }

    /**
     * Example 2: Setup a game with AI opponent
     */
    public static void exampleSetupGame() throws Exception {
        System.out.println("=== Example 2: Setup Game with AI ===\n");

        // Create players
        Player human = new Player("Human Player", 10);
        AIPlayer ai = new AIPlayer("Computer", 10, AIDifficulty.MEDIUM);

        // Add ships to human player
        Ship carrier = new Carrier();
        human.addShip(carrier);
        human.placeShipOnGrid(carrier, new Coordinate(0, 0), Direction.HORIZONTAL);

        Ship cruiser = new Cruiser();
        human.addShip(cruiser);
        human.placeShipOnGrid(cruiser, new Coordinate(0, 2), Direction.HORIZONTAL);

        System.out.println("Human ships: " + human.getShips().size() + " placed");

        // Add ships to AI player
        Ship aiCarrier = new Carrier();
        ai.addShip(aiCarrier);
        ai.placeShipOnGrid(aiCarrier, new Coordinate(5, 5), Direction.VERTICAL);

        Ship aiCruiser = new Cruiser();
        ai.addShip(aiCruiser);
        ai.placeShipOnGrid(aiCruiser, new Coordinate(7, 2), Direction.VERTICAL);

        System.out.println("AI ships: " + ai.getShips().size() + " placed");

        // Create game
        Game game = new Game(human, ai);
        game.start();

        System.out.println("Game started! Current player: " + game.getCurrentPlayer().getName());
        System.out.println();
    }

    /**
     * Example 3: Handle AI turn
     */
    public static void exampleAITurn() throws Exception {
        System.out.println("=== Example 3: AI Turn Handling ===\n");

        // Setup
        Player human = new Player("Human", 10);
        AIPlayer ai = new AIPlayer("AI", 10, AIDifficulty.HARD);

        // Add one ship to human
        Ship destroyer = new Destroyer();
        human.addShip(destroyer);
        human.placeShipOnGrid(destroyer, new Coordinate(3, 3), Direction.HORIZONTAL);

        // Add one ship to AI
        Ship aiDestroyer = new Destroyer();
        ai.addShip(aiDestroyer);
        ai.placeShipOnGrid(aiDestroyer, new Coordinate(6, 6), Direction.HORIZONTAL);

        Game game = new Game(human, ai);
        game.start();

        // Simulate a few turns
        for (int turn = 1; turn <= 5; turn++) {
            System.out.println("Turn " + turn + ":");

            Player current = game.getCurrentPlayer();

            if (current instanceof AIPlayer) {
                AIPlayer aiPlayer = (AIPlayer) current;

                // AI chooses where to shoot
                Coordinate shot = aiPlayer.chooseShot(game.getOpponent().getShips());
                System.out.println("  AI shoots at: " + shot);

                // Execute the attack
                AttackResponse response = game.playTurn(shot);

                // Notify AI of the result (important for learning!)
                aiPlayer.notifyAttackResult(shot, response);

                System.out.println("  Result: " + response.getResult());
            } else {
                // Human turn - simulate random shot
                Coordinate humanShot = new Coordinate(turn, turn);
                System.out.println("  Human shoots at: " + humanShot);

                AttackResponse response = game.playTurn(humanShot);
                System.out.println("  Result: " + response.getResult());
            }

            System.out.println();

            if (game.isGameOver()) {
                System.out.println("Game Over! Winner: " + game.getWinner().getName());
                break;
            }
        }
    }

    /**
     * Example 4: Test AI behavior at different difficulties
     */
    public static void exampleDifficultyComparison() {
        System.out.println("=== Example 4: Difficulty Comparison ===\n");

        AIPlayer easy = new AIPlayer("Easy", 10, AIDifficulty.EASY);
        AIPlayer medium = new AIPlayer("Medium", 10, AIDifficulty.MEDIUM);
        AIPlayer hard = new AIPlayer("Hard", 10, AIDifficulty.HARD);

        // Make 5 shots with each difficulty
        System.out.println("First 5 shots from each AI:\n");

        System.out.println("EASY AI:");
        for (int i = 0; i < 5; i++) {
            Coordinate shot = easy.chooseShot(easy.getShips());
            easy.getTrackingGrid().getCell(shot).shoot();
            System.out.println("  Shot " + (i + 1) + ": " + shot);
        }

        System.out.println("\nMEDIUM AI:");
        for (int i = 0; i < 5; i++) {
            Coordinate shot = medium.chooseShot(medium.getShips());
            medium.getTrackingGrid().getCell(shot).shoot();
            System.out.println("  Shot " + (i + 1) + ": " + shot);
        }

        System.out.println("\nHARD AI:");
        for (int i = 0; i < 5; i++) {
            Coordinate shot = hard.chooseShot(hard.getShips());
            hard.getTrackingGrid().getCell(shot).shoot();
            System.out.println("  Shot " + (i + 1) + ": " + shot);
        }

        System.out.println();
    }

    /**
     * Example 5: AI learning from hits
     */
    public static void exampleAILearning() {
        System.out.println("=== Example 5: AI Learning (Medium difficulty) ===\n");

        AIPlayer ai = new AIPlayer("Learning AI", 10, AIDifficulty.MEDIUM);

        // First shot (random)
        Coordinate shot1 = ai.chooseShot(ai.getShips());
        ai.getTrackingGrid().getCell(shot1).shoot();
        System.out.println("Shot 1 (HUNT mode): " + shot1);

        // Simulate a HIT
        Ship dummyShip = new Cruiser();
        AttackResponse hitResponse = new AttackResponse(
            com.par_28.ship_battle.model.enums.AttackResult.HIT,
            dummyShip
        );
        ai.notifyAttackResult(shot1, hitResponse);
        System.out.println("Result: HIT! AI switches to TARGET mode");

        // Next shots should be adjacent to the hit
        System.out.println("\nNext 4 shots (TARGET mode - should be adjacent to " + shot1 + "):");
        for (int i = 0; i < 4; i++) {
            Coordinate shot = ai.chooseShot(ai.getShips());
            ai.getTrackingGrid().getCell(shot).shoot();
            System.out.println("  Shot " + (i + 2) + ": " + shot);

            // Check if adjacent
            boolean isAdjacent = isAdjacentTo(shot, shot1);
            System.out.println("    Adjacent to hit? " + isAdjacent);
        }

        System.out.println();
    }

    /**
     * Helper method to check if two coordinates are adjacent
     */
    private static boolean isAdjacentTo(Coordinate c1, Coordinate c2) {
        int dx = Math.abs(c1.getX() - c2.getX());
        int dy = Math.abs(c1.getY() - c2.getY());
        return (dx == 1 && dy == 0) || (dx == 0 && dy == 1);
    }

    /**
     * Main method to run all examples
     */
    public static void main(String[] args) {
        try {
            exampleCreateAIPlayers();
            exampleSetupGame();
            exampleAITurn();
            exampleDifficultyComparison();
            exampleAILearning();

            System.out.println("=== All Examples Completed ===");
        } catch (Exception e) {
            System.err.println("Error running examples: " + e.getMessage());
            e.printStackTrace();
        }
    }
}
